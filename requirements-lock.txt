#   conda activate rag-energy
#   python -m pip freeze --exclude-editable > requirements-lock.txt

# Do not hand-write version numbers that have not been resolved locally. A lock
# file with invented pins is worse than no lock file: it fails silently on a
# different platform.
#

#   - OS and Python build
#   - GPU model, driver version, CUDA runtime
#   - Ollama version and model digest
